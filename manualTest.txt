# Acquire
curl -N "http://localhost:8080/api/v1/locks/acquire?lockName=order-100"

# Release
curl -X POST "http://localhost:8080/api/v1/locks/release" \
     -H "Content-Type: application/json" \
     -d '{"lockName": "order-100", "tokenId": "<TOKEN_ID>"}'

# Revoke
curl -X POST "http://localhost:8080/api/v1/locks/revoke" \
     -H "Content-Type: application/json" \
     -d '{"lockName": "order-100", "tokenId": "<TOKEN_ID>"}'


# Powershell curls :-

curl.exe -N "http://localhost:8080/api/v1/locks/acquire?lockName=order-100"

Invoke-RestMethod -Uri "http://localhost:8080/api/v1/locks/release" -Method Post -ContentType "application/json" -Body '{"lockName":"order-100","tokenId":"PASTE_TOKEN_HERE"}'

Invoke-RestMethod -Uri "http://localhost:8080/api/v1/locks/revoke" -Method Post -ContentType "application/json" -Body '{"lockName":"order-100","tokenId":"PASTE_TOKEN_HERE"}'
